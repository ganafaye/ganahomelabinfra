Méthode 1 : Installation Native (Bare Metal)

1. Installation des dépendances système

# Paquets de compilation
sudo apt install -y build-essential

# Bibliothèques de développement
sudo apt install -y \
    libxml2-dev \
    libxslt1-dev \
    libldap2-dev \
    libsasl2-dev \
    libssl-dev \
    libjpeg-dev \
    libpq-dev \
    libffi-dev \
    zlib1g-dev

# Outils supplémentaires
sudo apt install -y \
    git \
    python3-pip \
    python3-dev \
    python3-venv \
    nodejs \
    npm \
    xfonts-75dpi \
    xfonts-base

2. Installation et configuration de PostgreSQL

# Installation
sudo apt install -y postgresql postgresql-contrib

# Démarrage du service
sudo systemctl enable postgresql
sudo systemctl start postgresql

# Création de l'utilisateur Odoo
sudo su - postgres -c "createuser -s odoo18"

# Définition du mot de passe
sudo su - postgres -c "psql -c \"ALTER USER odoo18 WITH PASSWORD 'MotDePasseSecurise';\""

# Vérification
sudo systemctl status postgresql

3. Installation de wkhtmltopdf

Important : Utiliser la version 0.12.6 spécifiquement pour éviter les erreurs de rendu PDF. 
# Téléchargement de la version compatible
cd /tmp
wget https://github.com/wkhtmltopdf/packaging/releases/download/0.12.6.1-3/wkhtmltox_0.12.6.1-3.jammy_amd64.deb

# Installation
sudo dpkg -i wkhtmltox_0.12.6.1-3.jammy_amd64.deb

# Correction des dépendances si nécessaire
sudo apt --fix-broken install -y

# Vérification
wkhtmltopdf --version

4. Installation d'Odoo

# Création de l'utilisateur système Odoo
sudo useradd -m -d /opt/odoo18 -U -G sudo -s /bin/bash odoo

# Clonage du dépôt Odoo
sudo git clone https://www.github.com/odoo/odoo --depth 1 --branch 18.0 /opt/odoo18/odoo

# Attribution des permissions
sudo chown -R odoo:odoo /opt/odoo18

# Création du répertoire pour les modules personnalisés
sudo mkdir -p /opt/odoo18/custom-addons
sudo chown -R odoo:odoo /opt/odoo18/custom-addons

5. Configuration de l'environnement virtuel Python

# Passage à l'utilisateur odoo
sudo su - odoo

# Création de l'environnement virtuel
cd /opt/odoo18
python3 -m venv venv

# Activation de l'environnement
source venv/bin/activate

# Mise à jour de pip
pip install --upgrade pip

# Installation des dépendances Python
pip install wheel
pip install -r odoo/requirements.txt

# Désactivation de l'environnement
deactivate
exit

6. Fichier de configuration Odoo

# Création du répertoire de configuration
sudo mkdir -p /etc/odoo
sudo chown -R odoo:odoo /etc/odoo

# Création du fichier de configuration
sudo nano /etc/odoo/odoo.conf
[options]
; Mot de passe administrateur (à changer !)
admin_passwd = AdminOdoo2026!

; Base de données
db_host = localhost
db_port = 5432
db_user = odoo18
db_password = MotDePasseSecurise
db_name = False

; Chemins
addons_path = /opt/odoo18/odoo/addons,/opt/odoo18/custom-addons
data_dir = /opt/odoo18/data

; Logs
logfile = /var/log/odoo/odoo-server.log
log_level = info
logrotate = True

; Paramètres réseau
xmlrpc_port = 8069
longpolling_port = 8072
workers = 2
max_cron_threads = 1

; Paramètres de session
session_store = file
# Création des répertoires de logs et données
sudo mkdir -p /var/log/odoo
sudo chown -R odoo:odoo /var/log/odoo
sudo mkdir -p /opt/odoo18/data
sudo chown -R odoo:odoo /opt/odoo18/data

7. Création du service systemd

sudo nano /etc/systemd/system/odoo.service
[Unit]
Description=Odoo 18 ERP
Documentation=https://www.odoo.com/documentation/18.0/
Requires=postgresql.service
After=network.target postgresql.service

[Service]
Type=simple
SyslogIdentifier=odoo
PermissionsStartOnly=true
User=odoo
Group=odoo
ExecStart=/opt/odoo18/venv/bin/python3 /opt/odoo18/odoo/odoo-bin -c /etc/odoo/odoo.conf
WorkingDirectory=/opt/odoo18/odoo
StandardOutput=journal+console
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
# Activation et démarrage
sudo systemctl daemon-reload
sudo systemctl enable odoo.service
sudo systemctl start odoo.service

# Vérification
sudo systemctl status odoo.service
sudo journalctl -u odoo.service -f

Méthode 2 : Installation par Conteneurs (Docker)

Approche moderne : L'approche par conteneurs isole totalement l'application de l'infrastructure physique, facilitant les migrations et les sauvegardes.

1. Installation de Docker

# Installation des prérequis
sudo apt update
sudo apt install -y \
    apt-transport-https \
    ca-certificates \
    curl \
    gnupg \
    lsb-release

# Ajout de la clé GPG Docker
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg

# Ajout du dépôt Docker
echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu \
  $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# Installation de Docker
sudo apt update
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin

# Ajout de l'utilisateur au groupe docker
sudo usermod -aG docker $USER

# Activation et démarrage
sudo systemctl enable docker
sudo systemctl start docker

# Vérification
docker --version
docker compose version

# IMPORTANT : Se déconnecter et reconnecter
exit
# Se reconnecter ensuite

2. Création de la structure de projet

# Création du répertoire de travail
mkdir -p ~/odoo-docker/{addons,config,logs,backups}
cd ~/odoo-docker

# Création du fichier de configuration Odoo
nano config/odoo.conf
[options]
admin_passwd = AdminOdoo2026!
db_host = db
db_port = 5432
db_user = odoo
db_password = odoo2025!
addons_path = /mnt/extra-addons,/usr/lib/python3/dist-packages/odoo/addons
data_dir = /var/lib/odoo
logfile = /var/log/odoo/odoo-server.log
log_level = info

3. Fichier Docker Compose

nano docker-compose.yml
version: '3.8'

services:
  # Service PostgreSQL
  db:
    image: postgres:15
    container_name: odoo-db
    restart: always
    environment:
      - POSTGRES_DB=postgres
      - POSTGRES_USER=odoo
      - POSTGRES_PASSWORD=odoo2025!
      - PGDATA=/var/lib/postgresql/data/pgdata
    volumes:
      - db-data:/var/lib/postgresql/data
      - ./backups:/backups
    networks:
      - odoo-net
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U odoo"]
      interval: 10s
      timeout: 5s
      retries: 5

  # Service Redis (optionnel - pour cache/sessions)
  redis:
    image: redis:7-alpine
    container_name: odoo-redis
    restart: always
    volumes:
      - redis-data:/data
    networks:
      - odoo-net
    command: redis-server --appendonly yes

  # Service Odoo
  odoo:
    image: odoo:18.0
    container_name: odoo-app
    restart: always
    depends_on:
      db:
        condition: service_healthy
    ports:
      - "8069:8069"
      - "8072:8072"  # Longpolling
    environment:
      - HOST=db
      - USER=odoo
      - PASSWORD=odoo2025!
    volumes:
      - odoo-data:/var/lib/odoo
      - ./addons:/mnt/extra-addons:rw
      - ./config:/etc/odoo:ro
      - ./logs:/var/log/odoo
    networks:
      - odoo-net
    command: odoo
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8069"]
      interval: 30s
      timeout: 10s
      retries: 3

  # Service pgAdmin (optionnel - interface d'admin PostgreSQL)
  pgadmin:
    image: dpage/pgadmin4
    container_name: odoo-pgadmin
    restart: always
    depends_on:
      - db
    ports:
      - "5050:80"
    environment:
      - PGADMIN_DEFAULT_EMAIL=admin@odoo.local
      - PGADMIN_DEFAULT_PASSWORD=Admin2026!
    volumes:
      - pgadmin-data:/var/lib/pgadmin
    networks:
      - odoo-net

networks:
  odoo-net:
    driver: bridge

volumes:
  db-data:
  odoo-data:
  redis-data:
  pgadmin-data:

4. Lancement et gestion des conteneurs

# Se placer dans le répertoire du projet
cd ~/odoo-docker

# Lancement des services
docker compose up -d

# Vérification du statut
docker compose ps

# Visualisation des logs
docker compose logs -f

# Arrêt des services
docker compose down

# Arrêt avec suppression des volumes (⚠️ supprime les données)
docker compose down -v

# Redémarrage d'un service spécifique
docker compose restart odoo

# Mise à jour des images
docker compose pull

# Rebuild des images
docker compose build --no-cache

# Exécution de commandes dans le conteneur
docker exec -it odoo-app /bin/bash
docker exec -it odoo-db psql -U odoo -d postgres

5. Script d'automatisation

# Création d'un script de gestion
nano ~/odoo-docker/odoo-manager.sh
#!/bin/bash
# Script de gestion d'Odoo Docker

cd ~/odoo-docker

case "$1" in
    start)
        docker compose up -d
        echo "✅ Odoo démarré sur http://localhost:8069"
        ;;
    stop)
        docker compose down
        echo "✅ Odoo arrêté"
        ;;
    restart)
        docker compose restart
        echo "✅ Odoo redémarré"
        ;;
    logs)
        docker compose logs -f
        ;;
    backup)
        BACKUP_FILE="backup_$(date +%Y%m%d_%H%M%S).sql"
        docker exec -t odoo-db pg_dump -U odoo postgres > backups/$BACKUP_FILE
        echo "✅ Backup créé: backups/$BACKUP_FILE"
        ;;
    restore)
        if [ -z "$2" ]; then
            echo "Usage: $0 restore "
            exit 1
        fi
        cat backups/$2 | docker exec -i odoo-db psql -U odoo -d postgres
        echo "✅ Restauration effectuée"
        ;;
    shell)
        docker exec -it odoo-app /bin/bash
        ;;
    psql)
        docker exec -it odoo-db psql -U odoo -d postgres
        ;;
    clean)
        docker system prune -f
        echo "✅ Nettoyage effectué"
        ;;
    *)
        echo "Usage: $0 {start|stop|restart|logs|backup|restore|shell|psql|clean}"
        exit 1
esac
# Rendre le script exécutable
chmod +x ~/odoo-docker/odoo-manager.sh

# Utilisation
./odoo-manager.sh start
./odoo-manager.sh logs
./odoo-manager.sh backup

Sécurisation du serveur

1. Configuration du pare-feu (UFW)

# Installation si nécessaire
sudo apt install -y ufw

# Configuration de base
sudo ufw default deny incoming
sudo ufw default allow outgoing

# Ouverture des ports
sudo ufw allow 22/tcp comment 'SSH'
sudo ufw allow 80/tcp comment 'HTTP'
sudo ufw allow 443/tcp comment 'HTTPS'
sudo ufw allow 8069/tcp comment 'Odoo dev'

# Activation
sudo ufw --force enable

# Vérification
sudo ufw status verbose

2. Configuration de fail2ban

# Installation
sudo apt install -y fail2ban

# Configuration
sudo nano /etc/fail2ban/jail.local
[sshd]
enabled = true
port = ssh
filter = sshd
logpath = /var/log/auth.log
maxretry = 3
bantime = 3600

[odoo]
enabled = true
port = 8069
filter = odoo
logpath = /var/log/odoo/odoo-server.log
maxretry = 5
bantime = 1800
# Création du filtre Odoo
sudo nano /etc/fail2ban/filter.d/odoo.conf
[Definition]
failregex = ^.*openerp.*Login failed for login:.*$
ignoreregex =
# Redémarrage
sudo systemctl restart fail2ban
sudo systemctl status fail2ban

Maintenance et sauvegardes

Script de sauvegarde automatique

# Création du script
sudo nano /usr/local/bin/backup-odoo.sh
#!/bin/bash
# Script de sauvegarde automatique Odoo

# Variables
BACKUP_DIR="/opt/backups/odoo"
DATE=$(date +%Y%m%d_%H%M%S)
DB_NAME="homelab_odoo"
DB_USER="odoo18"
ODOO_DATA="/opt/odoo18/data"
RETENTION_DAYS=30

# Création du répertoire de backup
mkdir -p $BACKUP_DIR/{database,filestore}

# Sauvegarde de la base de données
echo "Sauvegarde de la base de données..."
pg_dump -U $DB_USER $DB_NAME | gzip > $BACKUP_DIR/database/odoo_db_$DATE.sql.gz

# Sauvegarde du filestore
echo "Sauvegarde des fichiers..."
tar -czf $BACKUP_DIR/filestore/odoo_filestore_$DATE.tar.gz -C $ODOO_DATA .

# Nettoyage des anciennes sauvegardes
echo "Nettoyage des sauvegardes de plus de $RETENTION_DAYS jours..."
find $BACKUP_DIR/database -type f -name "*.sql.gz" -mtime +$RETENTION_DAYS -delete
find $BACKUP_DIR/filestore -type f -name "*.tar.gz" -mtime +$RETENTION_DAYS -delete

echo "Sauvegarde terminée : $DATE"
# Rendre exécutable
sudo chmod +x /usr/local/bin/backup-odoo.sh

# Planification avec cron
sudo crontab -e
# Sauvegarde quotidienne à 2h du matin
0 2 * * * /usr/local/bin/backup-odoo.sh >> /var/log/odoo-backup.log 2>&1

# Nettoyage hebdomadaire des logs
0 3 * * 0 find /var/log/odoo -name "*.log" -mtime +30 -delete

Problème Cause possible Solution

  • Création de base possible
  • Installation de modules fonctionnelle
  • Génération de PDF fonctionnelle
